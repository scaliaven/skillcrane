# Benchmark environments for `python main.py --env <family>:<task>`.
# Install ON TOP of requirements.txt, ideally in a separate environment --
# these pin mujoco and numpy backwards from what the core project uses.
#
#   pip install -r requirements.txt -r requirements-benchmarks.txt
#
# Verified together on macOS 15 / Apple Silicon / Python 3.12:
#   robosuite 1.5.2  metaworld 3.1.1  gymnasium 1.3.0  gymnasium-robotics 1.4.2
#   mujoco 3.3.0     numpy 1.26.4
#
# NOT included: LIBERO. It pins robosuite==1.4.0, whose controller API differs,
# so it needs its own environment -- see environment-libero.yml.

# --- the hard constraint ----------------------------------------------------
# robosuite's mujoco-py compatibility shim does
#     assert joint_type in (mjtJoint.mjJNT_HINGE, mjtJoint.mjJNT_SLIDE)
# and in mujoco 3.12 the enum's *reflected* comparison with numpy.int32 started
# returning False, so that membership test fails and every robosuite env dies
# in _setup_references. Bisected: fine through 3.11.0, broken at 3.12.0.
mujoco<3.12

# numba (via robosuite) has no wheels for numpy 2.x yet, which is what drags
# numpy back to the 1.26 line. The core Claw Crew suite passes on both.
numpy<2

robosuite>=1.5,<1.6
metaworld>=3.1
gymnasium>=1.1
gymnasium-robotics>=1.4
